import torch
import timm

model = timm.create_model('efficientnet_b0', pretrained=False, num_classes=4)

저장된 모델 상태를 로드
model.load_state_dict(torch.load('TEST_MODEL_EFFICIENTNETB0.pth'))

평가 모드 설정
model.eval()

더미 입력 생성
dummy_input = torch.randn(1, 3, 224, 224)

모델 실행
pytorch_output = model(dummy_input)

출력 확인
print("PyTorch 모델 출력:", pytorch_output.detach().numpy())
